> preprocess_seattle <- function(df) {
+     names(df) <- tolower(names(df))
+     char_cols <- names(df)[sapply(df, is.character)]
+     fac_cols <- names(df)[sapply(df, is.factor)]
+     df <- df %>% mutate(across(all_of(char_cols), tolower)) %>% 
+         mutate(across(all_of(fac_cols), ~factor(tolower(as.character(.))))) %>% 
+         mutate(station = case_when(!is.na(station) & str_trim(station) == 
+             "seattle - beacon hill" ~ "beacon hill", !is.na(station) & 
+             str_trim(station) == "seattle-10th & weller" ~ "weller street", 
+             TRUE ~ station))
+     rename_map <- c(daily.mean.pm2.5.concentration = "pm25_mean", 
+         daily.mean.pm10.concentration = "pm10_mean", daily.max.1.hour.no2.concentration = "no2_max", 
+         daily.max.8.hour.co.concentration = "co_max", daily.max.8.hour.ozone.concentration = "ozone_max", 
+         daily.max.1.hour.so2.concentration = "so2_max", simple.assault = "si_assault", 
+         aggravated.assault = "ag_assault", theft.from.motor.vehicle = "theft_from_veh", 
+         motor.vehicle.theft = "theft_of_veh", burglary.breaking...entering = "burglary", 
+         all.other.larceny = "larceny", murder...nonnegligent.manslaughter = "murder", 
+         negligent.manslaughter = "manslaughter", smokepm_pred = "smokepm_pred", 
+         smokepm_dummy = "smokepm_dummy", wdf2 = "wdf2", wdf5 = "wdf5")
+     for (old in names(rename_map)) {
+         if (old %in% names(df)) 
+             names(df)[names(df) == old] <- rename_map[old]
+     }
+     factor_vars <- c("station", "year", "month", "dow", "week", 
+         "weekend", "holiday", "smokepm_dummy", "covid")
+     numeric_vars <- c("pm25_mean", "pm10_mean", "no2_max", "co_max", 
+         "ozone_max", "so2_max", "tmax", "tmin", "prcp", "si_assault", 
+         "ag_assault", "theft_from_veh", "theft_of_veh", "burglary", 
+         "larceny", "murder", "manslaughter", "temperature_1000", 
+         "temperature_975", "tcc", "blh", "dv_calls", "respt_mean", 
+         "servt_mean", "uof_l1", "uof_l2", "uof_ois", "uof_l3", 
+         "uof_total", "colsev0", "colsev1", "colsev2", "colsev2b", 
+         "colsev3", "totalsev", "cit_calls", "robbery", "intimidation", 
+         "person", "property", "rape", "smokepm_pred", "wdf2", 
+         "wdf5", "awnd")
+     df <- df %>% dplyr::mutate(dplyr::across(dplyr::any_of(factor_vars), 
+         ~as.factor(.))) %>% dplyr::mutate(dplyr::across(dplyr::any_of(numeric_vars), 
+         ~as.numeric(.)))
+     if ("date" %in% names(df)) {
+         df <- df %>% dplyr::mutate(date = as.Date(date)) %>% 
+             dplyr::mutate(month = factor(lubridate::month(date, 
+                 label = TRUE, abbr = FALSE)), year = as.factor(lubridate::year(date)), 
+                 week = as.factor(lubridate::isoweek(date)), week_of_year = as.factor(lubridate::week(date)), 
+                 dow = factor(lubridate::wday(date, label = TRUE, 
+                   abbr = FALSE)), dow_num = lubridate::wday(date, 
+                   label = FALSE, abbr = FALSE)) %>% dplyr::relocate(year, 
+             month, week, dow, dow_num, .after = date)
+     }
+     if ("dow_num" %in% names(df)) {
+         df <- df %>% dplyr::mutate(weekend = as.factor(ifelse(dow_num %in% 
+             c(1, 7), 1, 0)))
+     }
+     if ("year" %in% names(df)) {
+         df <- df %>% dplyr::mutate(covid = as.factor(ifelse(year %in% 
+             c(2020, 2021), 1, 0)))
+     }
+     if (all(c("station", "ozone_max", "date") %in% names(df))) {
+         df$station <- trimws(tolower(as.character(df$station)))
+         df$date <- as.Date(df$date)
+         beacon_ozone <- df %>% dplyr::filter(station == "beacon hill") %>% 
+             dplyr::select(date, ozone_max) %>% dplyr::rename(beacon_ozone = ozone_max)
+         df <- df %>% dplyr::left_join(beacon_ozone, by = "date") %>% 
+             dplyr::mutate(beacon_ozone = as.numeric(beacon_ozone), 
+                 ozone_max_i = dplyr::case_when(is.na(ozone_max) & 
+                   station == "weller street" ~ beacon_ozone, 
+                   TRUE ~ ozone_max)) %>% dplyr::select(-beacon_ozone)
+     }
+     for (col in c("tmin", "tmax")) {
+         if (col %in% names(df)) {
+             if (is.factor(df[[col]])) 
+                 df[[col]] <- as.character(df[[col]])
+             df[[col]] <- as.numeric(df[[col]])
+             df[[col]] <- convert_temperature(df[[col]], old_metric = "f", 
+                 new_metric = "c")
+         }
+     }
+     if ("date" %in% names(df)) 
+         df <- df %>% dplyr::mutate(year_month = format(date, 
+             "%Y-%m"))
+     if ("date" %in% names(df)) 
+         df <- df %>% dplyr::mutate(year_week = format(date, "%Y-%U"))
+     if (all(c("si_assault", "ag_assault") %in% names(df))) {
+         df$assault <- df$si_assault + df$ag_assault
+     }
+     if ("wdf2" %in% names(df)) {
+         df <- df %>% mutate(wind_8bin = cut(wdf2, breaks = c(0, 
+             22.5, 67.5, 112.5, 157.5, 202.5, 247.5, 292.5, 337.5, 
+             360), labels = c("N", "NE", "E", "SE", "S", "SW", 
+             "W", "NW", "N"), include.lowest = TRUE), wind_4bin = cut(wdf2, 
+             breaks = c(0, 45, 135, 225, 315, 360), labels = c("N", 
+                 "E", "S", "W", "N"), include.lowest = TRUE))
+         df$wind_8bin <- factor(df$wind_8bin)
+         df$wind_4bin <- factor(df$wind_4bin)
+     }
+     if ("tmax" %in% names(df)) {
+         df <- df %>% mutate(tmax_sq = tmax^2, tmin_sq = tmin^2)
+     }
+     if ("tmax" %in% names(df)) {
+         tmax_75 <- quantile(df$tmax, 0.75, na.rm = TRUE)
+         df <- df %>% mutate(high_temp = as.factor(ifelse(tmax >= 
+             tmax_75, 1, 0)))
+     }
+     if ("smokepm_pred" %in% names(df)) {
+         df <- df %>% mutate(smokepm_pred = ifelse(is.na(smokepm_pred), 
+             0, smokepm_pred))
+     }
+     log_vars <- c("si_assault", "ag_assault", "assault", "burglary", 
+         "robbery", "intimidation", "theft_from_veh", "theft_of_veh", 
+         "larceny", "murder", "person", "property", "rape")
+     for (v in log_vars) {
+         if (v %in% names(df)) {
+             df[[paste0("log_", v)]] <- log(1 + as.numeric(df[[v]]))
+         }
+     }
+     if ("date_x" %in% names(df)) 
+         df <- df %>% dplyr::select(-date_x)
+     grouped_cols <- list(a = c("dv_calls", "respt_mean", "servt_mean"), 
+         b = c("uof_l1", "uof_l2", "uof_ois", "uof_l3", "uof_total"), 
+         c = c("colsev0", "colsev1", "colsev2", "colsev2b", "colsev3", 
+             "totalsev"), d = c("cit_calls"))
+     for (group_cols in grouped_cols) {
+         present_cols <- intersect(group_cols, names(df))
+         if (length(present_cols) == 0) 
+             next
+         submat <- df[, present_cols, drop = FALSE]
+         mask <- apply(submat, 1, function(row) any(!is.na(row) & 
+             row != 0))
+         first_idx <- which(mask)[1]
+         if (!is.na(first_idx) && first_idx > 1) {
+             df[seq_len(first_idx - 1), present_cols] <- NA
+         }
+     }
+     return(df)
+ }

> fs_diag <- function(form_full, form_restr, data) {
+     fs <- feols(form_full, data = data, cluster = ~date)
+     fs0 <- feols(form_restr, data = data)
+     rss_f <- sum(residuals(fs)^2)
+     rss_0 <- sum(residuals(fs0)^2)
+     pr2 <- (rss_0 - rss_f)/rss_0
+     b <- coef(fs)[["z_smoke"]]
+     v <- vcov(fs)[["z_smoke", "z_smoke"]]
+     F <- if (is.na(b) || is.na(v)) 
+         NA
+     else as.numeric(b^2/v)
+     list(F = F, pr2 = pr2, fs = fs)
+ }

> run_ivglm <- function(form_fs_glm, form_ss_glm, data) {
+     data <- droplevels(data)
+     fs_glm <- glm(form_fs_glm, data = data)
+     ss_glm <- glm(form_ss_glm, family = poisson(link = "log"), 
+         data = data)
+     ivtools::ivglm(estmethod = "ts", fitX.LZ = fs_glm, fitY.LX = ss_glm, 
+         data = data, ctrl = TRUE, clusterid = "date", vcov.fit = TRUE)
+ }

> grab_by_name <- function(ivobj, vars) {
+     nm <- names(ivobj$est)
+     out <- lapply(vars, function(v) {
+         if (!(v %in% nm)) 
+             return(list(beta = NA, se = NA, p = NA))
+         b <- unname(ivobj$est[[v]])
+         se <- sqrt(ivobj$vcov[v, v])
+         p <- 2 * pnorm(-abs(b/se))
+         list(beta = b, se = se, p = p)
+     })
+     names(out) <- vars
+     out
+ }

> run_temp_2sri <- function(temp_controls, data) {
+     fs_formula <- as.formula(paste("pm25_mean ~ z_smoke + holiday + awnd + prcp +", 
+         temp_controls, "+ station + year + month + dow"))
+     fs_glm <- glm(fs_formula, data = data)

+             data = data, ctrl = TRUE, clusterid = "date", vcov.fit = TRUE)
+     }, error = function(e) NULL)
+     return(iv)

> run_week_2sri <- function(fe_spec, data, label) {
+     cat("Fitting:", label, "...\n")
+     fs_formula <- as.formula(paste("pm25_mean ~ z_smoke + holiday + awnd + tmax + tmin + prcp +", 
+         fe_spec))
+     ss_formula <- as.formula(paste("assault ~ pm25_mean + holiday + awnd + tmax + tmin + prcp +", 
+         fe_spec))
+     fs_glm <- glm(fs_formula, data = data)
+     ss_glm <- glm(ss_formula, family = poisson(link = "log"), 
+         data = data)
+     iv <- tryCatch({
+         ivtools::ivglm(estmethod = "ts", fitX.LZ = fs_glm, fitY.LX = ss_glm, 
+             data = data, ctrl = TRUE, clusterid = "date", vcov.fit = TRUE)
+     }, error = function(e) {
+         cat("  Error:", e$message, "\n")
+         NULL
+     })
+     fe_bars <- gsub(" \\+ ", " + ", fe_spec)
+     fe_bars <- gsub("station", "", fe_bars)
+     fe_bars <- gsub("^\\s*\\+\\s*", "", fe_bars)
+     fe_bars <- gsub("\\s*\\+\\s*$", "", fe_bars)
+     fe_bars <- paste("station +", fe_bars)
+     fs_fe <- tryCatch({
+         feols(as.formula(paste("pm25_mean ~ z_smoke + holiday + awnd + tmax + tmin + prcp |", 
+             fe_bars)), data = data, cluster = ~date)
+     }, error = function(e) NULL)
+     if (!is.null(fs_fe)) {
+         fs_b <- coef(fs_fe)["z_smoke"]
+         fs_se <- sqrt(vcov(fs_fe)["z_smoke", "z_smoke"])
+         F_stat <- (fs_b/fs_se)^2
+     }
+     else {
+         fs_b <- NA
+         fs_se <- NA
+         F_stat <- NA
+     }
+     if (!is.null(iv) && "pm25_mean" %in% names(iv$est)) {
+         b <- unname(iv$est[["pm25_mean"]])
+         se <- sqrt(iv$vcov["pm25_mean", "pm25_mean"])
+         p <- 2 * pnorm(-abs(b/se))
+     }
+     else {
+         b <- NA
+         se <- NA
+         p <- NA
+     }
+     list(b = b, se = se, p = p, fs_b = fs_b, fs_se = fs_se, F = F_stat, 
+         N = nrow(data))
+ }

> get_nl_results <- function(m, vars) {
+     coefs <- coef(m)
+     vcovs <- vcov(m)

> get_coef <- function(m, varname) {
+     if (!(varname %in% names(coef(m)))) {
+         return(list(b = NA, se = NA, p = NA))
+     }
+     b <- coef(m)[varname]
+     se <- sqrt(vcov(m)[varname, varname])
+     p <- 2 * pnorm(-abs(b/se))
+     list(b = b, se = se, p = p)
+ }

+     pct <- 100 * (exp(b) - 1)

> joint_wald_p <- function(mod, coef_names) {
+     bt <- coef(mod)
+     V <- vcov(mod)
+     keep <- intersect(coef_names, names(bt))
+     if (length(keep) == 0) 
+         return(NA)
+     b <- bt[keep]
+     Vsub <- V[keep, keep, drop = FALSE]
+     invV <- tryCatch(chol2inv(chol(Vsub)), error = function(e) MASS::ginv(as.matrix(Vsub)))
+     stat <- as.numeric(t(b) %*% invV %*% b)
+     pchisq(stat, df = length(keep), lower.tail = FALSE)
+ }
